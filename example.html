Quick Start

<!-- 1. Include the CSS and JS -->
<link rel="stylesheet" href="ambient-mixer.css">
<script src="ambient-mixer.js"></script>

<!-- 2. Create the mixer -->
<script>
const mixer = new AmbientMixer({
  masterVolume: 0.5,
  fadeSpeed: 0.02,
  moodDuration: { min: 20000, max: 40000 }
});

// 3. Define your audio tracks
mixer.init({
  night: '/audio/night-ambience.mp3',
  rain: '/audio/rain.mp3',
  thunder: '/audio/thunder.mp3',
  piano: '/audio/soft-piano.mp3',
  strings: '/audio/strings.mp3'
});

// 4. Define mood presets (track volume combinations)
mixer.defineMoods({
  calm: { night: 0.4, piano: 0.2 },
  rainy: { night: 0.3, rain: 0.4, piano: 0.15 },
  stormy: { night: 0.2, rain: 0.3, thunder: 0.4, strings: 0.1 },
  peaceful: { night: 0.35, piano: 0.25 },
  intense: { rain: 0.5, thunder: 0.4, strings: 0.2 }
});

// 5. Start playing
mixer.play();
</script>

API Reference

Methods

// Initialize with audio tracks
mixer.init({ trackName: '/path/to/audio.mp3', ... })

// Define mood presets
mixer.defineMoods({ moodName: { track1: 0.5, track2: 0.3 }, ... })

// Playback controls
mixer.play()           // Start the mixer
mixer.stop()           // Stop the mixer
mixer.toggle()         // Toggle play/stop

// Mood controls
mixer.setMood('calm')  // Set a specific mood
mixer.randomMood()     // Pick a random mood
mixer.enableAutoMood() // Enable automatic mood cycling
mixer.disableAutoMood()// Disable automatic mood cycling

// Volume
mixer.setMasterVolume(0.7)  // Set master volume (0-1)

// Info
mixer.getMoods()       // Get list of mood names
mixer.getCurrentMood() // Get current mood name
mixer.getIsPlaying()   // Check if playing

// Cleanup
mixer.destroy()        // Clean up resources

Configuration Options

new AmbientMixer({
  masterVolume: 0.5,      // Initial master volume (0-1)
  fadeSpeed: 0.02,        // How fast tracks fade in/out
  moodDuration: {
    min: 20000,           // Minimum ms before mood change
    max: 40000            // Maximum ms before mood change
  },
  onMoodChange: (mood) => {},      // Callback when mood changes
  onPlayStateChange: (playing) => {} // Callback when play state changes
})
